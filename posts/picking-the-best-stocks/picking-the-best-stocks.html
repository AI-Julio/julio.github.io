# 2) Clean 'Total Debt' and 'Total Cash' FIRST (before computing the ratio)
for col in ['Total Debt', 'Total Cash']:
    if col not in stocks_df.columns:
        raise KeyError(f"Column '{col}' not found in DataFrame.")
    s = stocks_df[col]
    # If it's object/string, strip $, % and commas, then convert to numeric
    if s.dtype == 'object':
        s = s.str.replace(r'[\$,%,]', '', regex=True)
    stocks_df[col] = pd.to_numeric(s, errors='coerce')

# 3) Compute Debt-to-Cash safely (avoid divide-by-zero -> NaN)
stocks_df['Debt-to-Cash'] = stocks_df['Total Debt'] / stocks_df['Total Cash'].replace({0: np.nan})
# Replace infinite values (if any) with NaN
stocks_df.replace([np.inf, -np.inf], np.nan, inplace=True)

# 4) Add 'Debt-to-Cash' to features after it exists
features.append('Debt-to-Cash')

  1. if you have any categories that needs rating this is the time to do that.

rating_map = {
    'Strong Buy': 5,

all_features = features + ['Top Rating Score', analyst_column]
normalized_df = selected_df.copy()
for feature in all_features:
    normalized_df[feature] = (
        selected_df[feature] - selected_df[feature].min()
    ) / (selected_df[feature].max() - selected_df[feature].min())
